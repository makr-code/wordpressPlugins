```php
<?php
/**
 * Data Encryption Service
 * 
 * Provides AES-256-GCM encryption/decryption for sensitive fields.
 * Supports key rotation and envelope encryption.
 */

class ThemisDB_Encryption {
    const ALGORITHM = 'aes-256-gcm';
    const NONCE_LENGTH = 12; // 96 bits for GCM
    const TAG_LENGTH = 16;   // 128 bits authentication tag
    
    private static $encryption_key = null;
    private static $key_version = 1;
    
    /**
     * Initialize encryption with WordPress secrets API (WP 6.2+)
     * Falls back to wp_salt() for older versions
     */
    public static function init() {
        if (function_exists('wp_get_default_encryption_key')) {
            // WordPress 6.2+ with secrets API
            self::$encryption_key = wp_get_default_encryption_key();
        } else {
            // Fallback for older WordPress
            self::$encryption_key = hash('sha256', 
                wp_salt('secure_auth') . wp_salt('logged_in'),
                false
            );
        }
        
        self::$key_version = get_option('themisdb_encryption_key_version', 1);
    }
    
    /**
     * Encrypt a string value
     * 
     * @param string $value Raw value to encrypt
     * @return string Encrypted value (base64 encoded with metadata)
     */
    public static function encrypt($value) {
        if (empty($value)) {
            return '';
        }
        
        // Generate random nonce for GCM
        $nonce = openssl_random_pseudo_bytes(self::NONCE_LENGTH, $strong);
        if (!$strong) {
            wp_die('System does not support strong random number generation');
        }
        
        // Encrypt with authentication tag
        $encrypted = openssl_encrypt(
            $value,
            self::ALGORITHM,
            self::$encryption_key,
            OPENSSL_RAW_DATA,
            $nonce,
            $tag
        );
        
        if ($encrypted === false) {
            throw new Exception('Encryption failed: ' . openssl_error_string());
        }
        
        // Pack: version + nonce + encrypted_data + auth_tag
        $packed = pack('N', self::$key_version)   // 4 bytes: key version
                . $nonce                           // 12 bytes: GCM nonce
                . $encrypted                       // variable: ciphertext
                . $tag;                            // 16 bytes: auth tag
        
        return base64_encode($packed);
    }
    
    /**
     * Decrypt a string value
     * 
     * @param string $encrypted Base64-encoded encrypted value
     * @return string Decrypted value
     */
    public static function decrypt($encrypted) {
        if (empty($encrypted)) {
            return '';
        }
        
        try {
            $packed = base64_decode($encrypted, true);
            if ($packed === false) {
                throw new Exception('Invalid base64 encoding');
            }
            
            // Unpack: version + nonce + encrypted_data + auth_tag
            $key_version = unpack('N', substr($packed, 0, 4))[1];
            $nonce = substr($packed, 4, self::NONCE_LENGTH);
            $tag = substr($packed, -self::TAG_LENGTH);
            $encrypted_data = substr($packed, 4 + self::NONCE_LENGTH, -self::TAG_LENGTH);
            
            // For now, only current key version supported
            // TODO: v2.0 - Support key rotation with version lookup
            if ($key_version !== self::$key_version) {
                throw new Exception('Encrypted data uses unsupported key version: ' . $key_version);
            }
            
            // Decrypt with authentication tag verification
            $decrypted = openssl_decrypt(
                $encrypted_data,
                self::ALGORITHM,
                self::$encryption_key,
                OPENSSL_RAW_DATA,
                $nonce,
                $tag
            );
            
            if ($decrypted === false) {
                throw new Exception('Decryption failed - authentication tag mismatch or corrupted data');
            }
            
            return $decrypted;
        } catch (Exception $e) {
            do_action('themisdb_encryption_error', $e->getMessage());
            throw $e;
        }
    }
    
    /**
     * Hash a value (one-way) for lookups
     * Used for customer email searches, etc.
     * 
     * @param string $value Value to hash
     * @return string SHA-256 hash (hex)
     */
    public static function hash($value) {
        return hash('sha256', $value . wp_salt('secure_auth'));
    }
}

// Auto-initialize on plugin load
add_action('init', ['ThemisDB_Encryption', 'init'], 1);
```

```php
<?php
/**
 * Validation Framework
 * 
 * Provides reusable validation rules and custom validators.
 */

class ThemisDB_Validator {
    private $rules = [];
    private $errors = [];
    private $data = [];
    
    public function __construct($data = []) {
        $this->data = $data;
    }
    
    /**
     * Add a validation rule
     * 
     * @param string $field Field name
     * @param string|array $rules Rule(s) to apply
     * @return $this
     */
    public function rule($field, $rules) {
        if (!isset($this->rules[$field])) {
            $this->rules[$field] = [];
        }
        
        if (is_string($rules)) {
            $this->rules[$field][] = $rules;
        } else {
            $this->rules[$field] = array_merge($this->rules[$field], (array)$rules);
        }
        
        return $this;
    }
    
    /**
     * Validate data
     * 
     * @return bool True if all validations pass
     */
    public function validate() {
        $this->errors = [];
        
        foreach ($this->rules as $field => $rules) {
            $value = $this->data[$field] ?? null;
            
            foreach ((array)$rules as $rule) {
                $result = $this->apply_rule($field, $value, $rule);
                if ($result !== true) {
                    if (!isset($this->errors[$field])) {
                        $this->errors[$field] = [];
                    }
                    $this->errors[$field][] = $result;
                }
            }
        }
        
        return empty($this->errors);
    }
    
    /**
     * Get validation errors
     * 
     * @return array Errors by field
     */
    public function errors() {
        return $this->errors;
    }
    
    /**
     * Get single field error
     * 
     * @param string $field Field name
     * @return array|null Errors for field or null
     */
    public function error($field) {
        return $this->errors[$field] ?? null;
    }
    
    /**
     * Apply a single rule
     * 
     * @param string $field Field name
     * @param mixed $value Field value
     * @param string $rule Rule name or pattern
     * @return true|string True if valid, error message if invalid
     */
    private function apply_rule($field, $value, $rule) {
        // Parse rule with parameters: "rule:param1,param2"
        $parts = explode(':', $rule, 2);
        $rule_name = $parts[0];
        $params = isset($parts[1]) ? explode(',', $parts[1]) : [];
        
        switch ($rule_name) {
            case 'required':
                return empty($value) ? 'This field is required' : true;
            
            case 'email':
                return !is_email($value) ? 'Invalid email address' : true;
            
            case 'url':
                return !filter_var($value, FILTER_VALIDATE_URL) ? 'Invalid URL' : true;
            
            case 'integer':
                return !is_numeric($value) || intval($value) != $value ? 'Must be an integer' : true;
            
            case 'numeric':
                return !is_numeric($value) ? 'Must be a number' : true;
            
            case 'min':
                $min = $params[0] ?? 0;
                return strlen($value) < $min ? "Minimum length is $min characters" : true;
            
            case 'max':
                $max = $params[0] ?? 255;
                return strlen($value) > $max ? "Maximum length is $max characters" : true;
            
            case 'regex':
                $pattern = $params[0] ?? '';
                return !preg_match($pattern, $value) ? 'Invalid format' : true;
            
            case 'in':
                return !in_array($value, $params) ? 'Invalid selection' : true;
            
            case 'unique':
                $table = $params[0] ?? '';
                $column = $params[1] ?? $field;
                return $this->is_unique($table, $column, $value) ? true : 'This value already exists';
            
            default:
                return true;
        }
    }
    
    /**
     * Check if value is unique in database
     * 
     * @param string $table Database table name
     * @param string $column Column name
     * @param mixed $value Value to check
     * @return bool True if unique
     */
    private function is_unique($table, $column, $value) {
        global $wpdb;
        
        $result = $wpdb->get_var(
            $wpdb->prepare(
                "SELECT COUNT(*) FROM $wpdb->prefix$table WHERE $column = %s",
                $value
            )
        );
        
        return $result == 0;
    }
}

// Convenience function
function themi db_validate($data, $rules) {
    $validator = new ThemisDB_Validator($data);
    
    foreach ($rules as $field => $field_rules) {
        $validator->rule($field, $field_rules);
    }
    
    return $validator->validate() ? true : $validator->errors();
}
```

```php
<?php
/**
 * Rate Limiting Service
 * 
 * Implements token bucket algorithm for rate limiting.
 * Stores limits in WordPress transients.
 */

class ThemisDB_Rate_Limiter {
    const DEFAULT_LIMIT = 60;      // Requests per window
    const DEFAULT_WINDOW = 3600;   // Time window in seconds (1 hour)
    
    /**
     * Check if request is allowed
     * 
     * @param string $key Rate limit key (e.g., 'api.order.create:user_123')
     * @param int $limit Max requests allowed
     * @param int $window Time window in seconds
     * @return bool True if request is allowed
     */
    public static function allow($key, $limit = self::DEFAULT_LIMIT, $window = self::DEFAULT_WINDOW) {
        $transient_key = 'themisdb_ratelimit_' . md5($key);
        $current = (int) get_transient($transient_key) ?: 0;
        
        if ($current >= $limit) {
            return false;
        }
        
        // Increment counter
        set_transient($transient_key, $current + 1, $window);
        
        return true;
    }
    
    /**
     * Get remaining requests
     * 
     * @param string $key Rate limit key
     * @param int $limit Max requests allowed
     * @return int Remaining requests
     */
    public static function remaining($key, $limit = self::DEFAULT_LIMIT) {
        $transient_key = 'themisdb_ratelimit_' . md5($key);
        $current = (int) get_transient($transient_key) ?: 0;
        
        return max(0, $limit - $current);
    }
    
    /**
     * Reset rate limit counter
     * 
     * @param string $key Rate limit key
     */
    public static function reset($key) {
        $transient_key = 'themisdb_ratelimit_' . md5($key);
        delete_transient($transient_key);
    }
}
```

**File:** `includes/core/class-audit-logger.php` (NEW)

```php
<?php
/**
 * Audit Logger Service
 * 
 * Immutable, tamper-evident logging for compliance.
 * Stores in dedicated wp_themisdb_audit_log table.
 */

class ThemisDB_Audit_Logger {
    const MAX_LOG_AGE = 90 * DAY_IN_SECONDS; // 90-day retention
    
    /**
     * Log an audit event
     * 
     * @param string $event_type Event type (e.g., 'order.created')
     * @param array $context Event context/metadata
     * @return int Log entry ID
     */
    public static function log($event_type, $context = []) {
        global $wpdb;
        
        $table_name = $wpdb->prefix . 'themisdb_audit_log';
        
        $entry = [
            'event_type' => $event_type,
            'actor_id' => get_current_user_id(),
            'actor_email' => wp_get_current_user()->user_email,
            'actor_ip' => self::get_client_ip(),
            'actor_user_agent' => $_SERVER['HTTP_USER_AGENT'] ?? 'unknown',
            'resource_type' => $context['resource_type'] ?? null,
            'resource_id' => $context['resource_id'] ?? null,
            'action_data' => wp_json_encode($context),
            'timestamp' => current_time('mysql', true),
            'hash' => null, // Will be calculated below
        ];
        
        // Calculate tamper-evident hash
        $entry['hash'] = self::calculate_hash($entry);
        
        // Insert into database
        $wpdb->insert(
            $table_name,
            [
                'event_type' => $entry['event_type'],
                'actor_id' => $entry['actor_id'],
                'actor_email' => $entry['actor_email'],
                'actor_ip' => $entry['actor_ip'],
                'actor_user_agent' => $entry['actor_user_agent'],
                'resource_type' => $entry['resource_type'],
                'resource_id' => $entry['resource_id'],
                'action_data' => $entry['action_data'],
                'timestamp' => $entry['timestamp'],
                'hash' => $entry['hash'],
            ],
            ['%s', '%d', '%s', '%s', '%s', '%s', '%d', '%s', '%s', '%s']
        );
        
        return $wpdb->insert_id;
    }
    
    /**
     * Query audit log
     * 
     * @param array $filters Filters (event_type, actor_id, resource_type, etc.)
     * @param array $options Query options (limit, offset, order)
     * @return array Log entries
     */
    public static function query($filters = [], $options = []) {
        global $wpdb;
        
        $table_name = $wpdb->prefix . 'themisdb_audit_log';
        $limit = $options['limit'] ?? 100;
        $offset = $options['offset'] ?? 0;
        $order = $options['order'] ?? 'DESC';
        
        $where_parts = [];
        $params = [];
        
        if (isset($filters['event_type'])) {
            $where_parts[] = 'event_type = %s';
            $params[] = $filters['event_type'];
        }
        if (isset($filters['actor_id'])) {
            $where_parts[] = 'actor_id = %d';
            $params[] = $filters['actor_id'];
        }
        if (isset($filters['resource_type'])) {
            $where_parts[] = 'resource_type = %s';
            $params[] = $filters['resource_type'];
        }
        if (isset($filters['resource_id'])) {
            $where_parts[] = 'resource_id = %d';
            $params[] = $filters['resource_id'];
        }
        
        $where_clause = !empty($where_parts) ? 'WHERE ' . implode(' AND ', $where_parts) : '';
        
        $query = $wpdb->prepare(
            "SELECT * FROM $table_name $where_clause ORDER BY id $order LIMIT %d OFFSET %d",
            array_merge($params, [$limit, $offset])
        );
        
        return $wpdb->get_results($query);
    }
    
    /**
     * Verify log integrity
     * 
     * @param int $log_id Log entry ID
     * @return bool True if entry has not been tampered with
     */
    public static function verify_integrity($log_id) {
        global $wpdb;
        
        $table_name = $wpdb->prefix . 'themisdb_audit_log';
        $entry = $wpdb->get_row(
            $wpdb->prepare("SELECT * FROM $table_name WHERE id = %d", $log_id)
        );
        
        if (!$entry) {
            return false;
        }
        
        $stored_hash = $entry->hash;
        $entry->hash = null;
        $calculated_hash = self::calculate_hash((array)$entry);
        
        return $stored_hash === $calculated_hash;
    }
    
    /**
     * Calculate tamper-evident hash
     * 
     * @param array $entry Log entry
     * @return string SHA-256 hash
     */
    private static function calculate_hash($entry) {
        $data = implode('|', [
            $entry['event_type'],
            $entry['actor_id'],
            $entry['actor_email'],
            $entry['actor_ip'],
            $entry['resource_type'],
            $entry['resource_id'],
            $entry['action_data'],
            $entry['timestamp'],
        ]);
        
        return hash('sha256', $data . wp_salt('secure_auth'));
    }
    
    /**
     * Get client IP address
     * 
     * @return string Client IP
     */
    private static function get_client_ip() {
        if (!empty($_SERVER['HTTP_CLIENT_IP'])) {
            return $_SERVER['HTTP_CLIENT_IP'];
        } elseif (!empty($_SERVER['HTTP_X_FORWARDED_FOR'])) {
            return explode(',', $_SERVER['HTTP_X_FORWARDED_FOR'])[0];
        } else {
            return $_SERVER['REMOTE_ADDR'] ?? 'unknown';
        }
    }
    
    /**
     * Clean up old log entries (automated via wp-cron)
     */
    public static function cleanup() {
        global $wpdb;
        
        $table_name = $wpdb->prefix . 'themisdb_audit_log';
        $cutoff_date = gmdate('Y-m-d H:i:s', time() - self::MAX_LOG_AGE);
        
        $wpdb->query(
            $wpdb->prepare(
                "DELETE FROM $table_name WHERE timestamp < %s",
                $cutoff_date
            )
        );
    }
}

// Auto-cleanup via cron
add_action('themisdb_audit_log_cleanup', ['ThemisDB_Audit_Logger', 'cleanup']);
```

#### 5.1 Security Headers Manager

**File:** `includes/security/class-security-headers.php` (NEW)

```php
<?php
/**
 * Security Headers Manager
 * 
 * Adds security headers to all plugin responses.
 */

class ThemisDB_Security_Headers {
    public static function init() {
        add_action('wp_headers', [self::class, 'add_security_headers']);
    }
    
    public static function add_security_headers($headers) {
        // HSTS - Enforce HTTPS for 1 year
        if (is_ssl()) {
            $headers['Strict-Transport-Security'] = 'max-age=31536000; includeSubDomains; preload';
        }
        
        // CSP - Content Security Policy
        $headers['Content-Security-Policy'] = self::get_csp_policy();
        
        // X-Frame-Options - Clickjacking protection
        $headers['X-Frame-Options'] = 'SAMEORIGIN';
        
        // X-Content-Type-Options - Prevent MIME type sniffing
        $headers['X-Content-Type-Options'] = 'nosniff';
        
        // X-XSS-Protection - XSS filter
        $headers['X-XSS-Protection'] = '1; mode=block';
        
        // Referrer-Policy
        $headers['Referrer-Policy'] = 'strict-origin-when-cross-origin';
        
        // Permissions-Policy (formerly Feature-Policy)
        $headers['Permissions-Policy'] = 'camera=(), microphone=(), geolocation=()';
        
        return $headers;
    }
    
    private static function get_csp_policy() {
        $nonce = wp_create_nonce('themisdb_csp_nonce');
        
        return "
            default-src 'self';
            script-src 'self' 'nonce-$nonce' cdnjs.cloudflare.com;
            style-src 'self' 'unsafe-inline' fonts.googleapis.com;
            font-src 'self' fonts.gstatic.com;
            img-src 'self' data: https:;
            frame-ancestors 'self';
            base-uri 'self';
            form-action 'self';
            upgrade-insecure-requests;
        ";
    }
}

add_action('init', ['ThemisDB_Security_Headers', 'init']);
```
